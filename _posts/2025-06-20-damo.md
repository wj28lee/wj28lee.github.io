```bash
$ git clone https://github.com/awslabs/damo.git
$ cd damo

# Case 1
$ ./damo record "<your application> <options>"
$ ./damo report heats --heatmap heatmap.pdf

# Case 2
$ ./damo record --ops -r "0-1073741824" & # 0x0-0x40000000
$ DAMO_PID=$!
$ ./<your application> <options>
$ kill -2 $DAMO_PID
$ sleep 10 # Wait until perf data is written. It could be 20s or more.
$ ./damo report heats --heatmap heatmap.pdf
```
